            localStorage.setItem('wh_cart_data', JSON.stringify(cart));

            if (confirm("Reset store data back to default demo items?")) {

                cart = [];
                saveStore();
                saveCart();

            document.getElementById('store-name').textContent = store.name;
            document.getElementById('store-greeting').textContent = store.greeting;
            document.getElementById('store-logo').textContent = store.name.charAt(0).toUpperCase();
        }

        function toggleViewMode() {
            isAdmin = !isAdmin;
            const customerView = document.getElementById('customer-view');
            const adminView = document.getElementById('admin-view');
            const floatingCart = document.getElementById('floating-cart-bar');
            const viewLabel = document.getElementById('view-mode-label');

                customerView.classList.add('hidden');
                adminView.classList.remove('hidden');
                floatingCart.classList.add('hidden');
                resetBtn.classList.remove('hidden');
                viewLabel.textContent = "Exit Admin";
                renderAdminView();
            } else {
                customerView.classList.remove('hidden');
                adminView.classList.add('hidden');
                floatingCart.classList.remove('hidden');
                resetBtn.classList.add('hidden');
                viewLabel.textContent = "Admin";
                renderCustomerMenu();
            }
        }

        // CATEGORIES
        function renderCategoryPills() {
            const container = document.getElementById('category-pills');
            container.innerHTML = store.categories.map(cat => `
                <button onclick="setCategory('${cat}')" class="px-4 py-2 rounded-full text-sm font-medium whitespace-nowrap transition ${

                }">
                    ${cat}
                </button>
            `).join('');
        }

        function setCategory(cat) {
            activeCategory = cat;
            renderCategoryPills();
            renderCustomerMenu();
        }

        // CUSTOMER MENU RENDER
        function renderCustomerMenu() {
            const container = document.getElementById('menu-items-container');
            const searchKeyword = document.getElementById('search-input').value.toLowerCase();

            const filteredItems = store.items.filter(item => {
                const matchesCategory = activeCategory === "All" || item.category === activeCategory;
                const matchesSearch = item.name.toLowerCase().includes(searchKeyword) || item.description.toLowerCase().includes(searchKeyword);
                return matchesCategory && matchesSearch;
            });

            if (filteredItems.length === 0) {
                container.innerHTML = `
                    <div class="col-span-full py-12 text-center text-slate-400">
                        <i class="fa-solid fa-utensils text-4xl mb-2"></i>
                        <p>No dishes found matching your criteria.</p>
                    </div>`;
                return;
            }

            container.innerHTML = filteredItems.map(item => `
                <div class="bg-white rounded-2xl overflow-hidden border border-slate-200 shadow-sm hover:shadow-md transition flex flex-col justify-between">
                    <div class="flex p-3 gap-3">
                        ${item.image ? `<img src="${item.image}" class="w-24 h-24 object-cover rounded-xl flex-shrink-0" alt="${item.name}">` : ''}
                        <div class="flex-1 min-w-0">
                            <h3 class="font-bold text-slate-900 text-base leading-snug truncate">${item.name}</h3>
                            <p class="text-xs text-slate-500 line-clamp-2 mt-1">${item.description}</p>
                            <div class="font-bold text-emerald-600 text-base mt-2">${store.currency}${item.price.toFixed(2)}</div>
                        </div>
                    </div>
                    <div class="px-3 pb-3 pt-0">
                        <button onclick="openCustomerCustomizationModal('${item.id}')" class="w-full py-2 bg-slate-100 hover:bg-emerald-50 hover:text-emerald-700 text-slate-700 text-sm font-semibold rounded-xl border border-slate-200 flex items-center justify-center gap-1.5 transition">
                            <i class="fa-solid fa-plus text-xs"></i> Add to Order
                        </button>
                    </div>
                </div>
            `).join('');
        }

        // CUSTOMER CUSTOMIZATION & CART

            if (!item) return;

            const modal = document.getElementById('item-modal-overlay');
            const content = document.getElementById('modal-content');

            content.innerHTML = `
                <div class="space-y-4">
                    <div class="flex justify-between items-start">
                        <div>
                            <h3 class="font-bold text-slate-900 text-xl">${item.name}</h3>
                            <p class="text-sm text-slate-500">${item.description}</p>
                        </div>
                        <button onclick="closeModal()" class="text-slate-400 hover:text-slate-600"><i class="fa-solid fa-xmark text-xl"></i></button>
                    </div>

                    ${item.image ? `<img src="${item.image}" class="w-full h-44 object-cover rounded-xl" alt="${item.name}">` : ''}

                    <form id="customization-form" class="space-y-4 pt-2">
                        ${(item.optionGroups || []).map((grp, gIdx) => `
                            <div class="border-t pt-3">
                                <label class="block font-bold text-sm text-slate-800 mb-2">${grp.name}</label>
                                <div class="space-y-2">
                                    ${grp.options.map((opt, oIdx) => `
                                        <label class="flex justify-between items-center p-2.5 rounded-xl border border-slate-200 hover:bg-slate-50 cursor-pointer text-sm">
                                            <div class="flex items-center gap-2">
                                                <input type="${grp.type === 'single' ? 'radio' : 'checkbox'}" 
                                                       name="grp_${gIdx}" 
                                                       value="${opt.name}" 
                                                       data-price="${opt.price}" 
                                                       ${grp.type === 'single' && oIdx === 0 ? 'checked' : ''}
                                                       class="text-emerald-600 focus:ring-emerald-500">
                                                <span class="text-slate-700">${opt.name}</span>
                                            </div>
                                            <span class="text-xs font-semibold text-slate-500">+${store.currency}${opt.price.toFixed(2)}</span>
                                        </label>
                                    `).join('')}
                                </div>
                            </div>
                        `).join('')}

                            <label class="block font-bold text-sm text-slate-800 mb-1">Special Instructions</label>

        function adjustModalQty(delta) {
            const qtySpan = document.getElementById('modal-qty');
            let qty = parseInt(qtySpan.textContent) + delta;
            if (qty < 1) qty = 1;
            qtySpan.textContent = qty;
        }

        function closeModal() {
            document.getElementById('item-modal-overlay').classList.add('hidden');
        }

        function addToCart(itemId) {
            const item = store.items.find(i => i.id === itemId);
            const qty = parseInt(document.getElementById('modal-qty').textContent);
            const notes = document.getElementById('special-instructions').value.trim();

            const selectedOptions = [];
            let optionsTotalPrice = 0;

            const form = document.getElementById('customization-form');
            const inputs = form.querySelectorAll('input[type="radio"]:checked, input[type="checkbox"]:checked');

            inputs.forEach(input => {
                const optPrice = parseFloat(input.dataset.price) || 0;
                selectedOptions.push({ name: input.value, price: optPrice });
                optionsTotalPrice += optPrice;
            });

            const cartItem = {
                cartItemId: Date.now().toString(),
                id: item.id,
                name: item.name,
                unitPrice: item.price + optionsTotalPrice,
                qty: qty,
                options: selectedOptions,
                notes: notes
            };

            cart.push(cartItem);
            saveCart();
            closeModal();
            toggleCartDrawer(true);
        }

        function updateCartBadges() {
            const totalQty = cart.reduce((sum, item) => sum + item.qty, 0);
            const totalPrice = cart.reduce((sum, item) => sum + (item.unitPrice * item.qty), 0);

            document.getElementById('cart-count-badge').textContent = totalQty;
            document.getElementById('cart-total-badge').textContent = `${store.currency}${totalPrice.toFixed(2)}`;
            document.getElementById('cart-drawer-total').textContent = `${store.currency}${totalPrice.toFixed(2)}`;
        }

        function toggleCartDrawer(open) {
            const drawer = document.getElementById('cart-drawer-overlay');
            if (open) {
                renderCartItems();
                drawer.classList.remove('hidden');
            } else {
                drawer.classList.add('hidden');
            }
        }

        function renderCartItems() {
            const container = document.getElementById('cart-items-list');

            if (cart.length === 0) {
                container.innerHTML = `
                    <div class="py-12 text-center text-slate-400">
                        <i class="fa-solid fa-basket-shopping text-4xl mb-2"></i>
                        <p>Your order is empty.</p>
                    </div>`;
                return;
            }

            container.innerHTML = cart.map((ci, index) => `
                <div class="pt-3 first:pt-0">
                    <div class="flex justify-between items-start">
                        <div>
                            <h4 class="font-bold text-slate-800 text-sm">${ci.name}</h4>
                            <div class="text-xs text-slate-500">
                                ${ci.options.map(o => `+ ${o.name}`).join(', ')}
                                ${ci.notes ? `<div class="italic text-slate-400">Note: ${ci.notes}</div>` : ''}
                            </div>
                        </div>
                        <span class="font-bold text-slate-900 text-sm">${store.currency}${(ci.unitPrice * ci.qty).toFixed(2)}</span>
                    </div>

                    <div class="flex items-center justify-between mt-2">
                        <div class="flex items-center border rounded-lg">
                            <button onclick="updateCartQty(${index}, -1)" class="px-2 py-0.5 text-slate-600 hover:bg-slate-100 font-bold text-xs">-</button>
                            <span class="px-2 text-xs font-semibold">${ci.qty}</span>
                            <button onclick="updateCartQty(${index}, 1)" class="px-2 py-0.5 text-slate-600 hover:bg-slate-100 font-bold text-xs">+</button>
                        </div>
                        <button onclick="removeCartItem(${index})" class="text-xs text-red-500 hover:underline">Remove</button>
                    </div>
                </div>
            `).join('');
        }

        function updateCartQty(index, delta) {
            cart[index].qty += delta;
            if (cart[index].qty <= 0) {
                cart.splice(index, 1);
            }
            saveCart();
            renderCartItems();
        }

        function removeCartItem(index) {
            cart.splice(index, 1);
            saveCart();

        }

        // CHECKOUT & WHATSAPP GENERATION
        function sendOrderViaWhatsApp() {
            if (cart.length === 0) {
                alert("Please add items to your order first!");
                return;
            }

            const note = document.getElementById('checkout-note').value.trim();
            const totalPrice = cart.reduce((sum, item) => sum + (item.unitPrice * item.qty), 0);

            let msg = `*NEW ORDER - ${store.name}*
`;
            msg += `------------------------------
`;

            cart.forEach(item => {
                msg += `• *${item.qty}x ${item.name}* - ${store.currency}${(item.unitPrice * item.qty).toFixed(2)}
`;
                if (item.options.length > 0) {
                    msg += `   Opts: ${item.options.map(o => o.name).join(', ')}
`;
                }
                if (item.notes) {
                    msg += `   Note: ${item.notes}
`;
                }
            });

            msg += `------------------------------
`;
            msg += `*Total Amount:* ${store.currency}${totalPrice.toFixed(2)}
`;
            if (note) {
                msg += `*Table / Address:* ${note}
`;
            }

            const encodedMsg = encodeURIComponent(msg);
            const targetPhone = store.phone.replace(/[^0-9]/g, '');
            const whatsappUrl = `https://wa.me/${targetPhone}?text=${encodedMsg}`;

            window.open(whatsappUrl, '_blank');
        }

        function switchAdminTab(tab) {
            currentAdminTab = tab;
            ['items', 'categories', 'settings'].forEach(t => {
                document.getElementById(`admin-tab-${t}`).classList.toggle('hidden', t !== tab);
                const btn = document.getElementById(`tab-btn-${t}`);
                if (t === tab) {
                    btn.className = "py-2 text-emerald-600 border-b-2 border-emerald-600 font-semibold";
                } else {
                    btn.className = "py-2 hover:text-slate-700";
                }
            });

            if (tab === 'settings') loadAdminSettings();
        }

        function saveMenuItem(e, itemId) {
            e.preventDefault();
            const name = document.getElementById('admin-item-name').value.trim();
            const price = parseFloat(document.getElementById('admin-item-price').value);
            const category = document.getElementById('admin-item-category').value;
            const description = document.getElementById('admin-item-desc').value.trim();
            const image = document.getElementById('admin-item-image').value.trim();

            const existingIndex = store.items.findIndex(i => i.id === itemId);
            const itemObj = {
                id: itemId,
                name,
                price,
                category,
                description,
                image,
                optionGroups: existingIndex >= 0 ? store.items[existingIndex].optionGroups : []
            };

            if (existingIndex >= 0) {
                store.items[existingIndex] = itemObj;
            } else {
                store.items.push(itemObj);
            }

            saveStore();
            closeModal();
            renderAdminItems();
        }

        function deleteMenuItem(itemId) {
            if (confirm("Are you sure you want to delete this menu item?")) {
                store.items = store.items.filter(i => i.id !== itemId);
                saveStore();
                renderAdminItems();
            }
        }

        // ADMIN CATEGORIES
        function renderAdminCategories() {
            const container = document.getElementById('admin-categories-list');
            container.innerHTML = store.categories.filter(c => c !== "All").map(cat => `
                <li class="p-3 flex justify-between items-center text-sm font-medium text-slate-700">
                    <span>${cat}</span>
                    <button onclick="deleteCategory('${cat}')" class="text-red-500 hover:underline text-xs">Remove</button>
                </li>
            `).join('');
        }

        function addCategory() {
            const input = document.getElementById('new-category-input');
            const val = input.value.trim();
            if (val && !store.categories.includes(val)) {
                store.categories.push(val);
                saveStore();
                input.value = '';
                renderAdminCategories();
                renderCategoryPills();
            }
        }

        function deleteCategory(cat) {
            if (confirm(`Delete category "${cat}"?`)) {
                store.categories = store.categories.filter(c => c !== cat);
                saveStore();
                renderAdminCategories();
                renderCategoryPills();
            }
        }

            store.greeting = document.getElementById('setting-greeting').value.trim() || store.greeting;
