    <script>
        const DEFAULT_STORE = {
            name: "Gourmet House",
            greeting: "Fresh & Artisanal Dishes",
            phone: "1234567890",
            currency: "$",
            categories: ["All", "Burgers", "Pizzas", "Drinks", "Desserts"],
            items: [
                {
                    id: "1",
                    name: "Classic Cheeseburger",
                    description: "Juicy beef patty, cheddar cheese, crisp lettuce, tomato & special sauce.",
                    price: 9.99,
                    category: "Burgers",
                    image: "https://images.unsplash.com/photo-1568901346375-23c9450c58cd?w=400&auto=format&fit=crop&q=80",
                    optionGroups: []
                },
                {
                    id: "2",
                    name: "Margherita Pizza",
                    description: "San Marzano tomatoes, fresh mozzarella, basil, and olive oil.",
                    price: 12.50,
                    category: "Pizzas",
                    image: "https://images.unsplash.com/photo-1604382354936-07c5d9983bd3?w=400&auto=format&fit=crop&q=80",
                    optionGroups: []
                }
            ]
        };

        let store = JSON.parse(localStorage.getItem('wh_store_data')) || DEFAULT_STORE;

        function authenticateAdmin() {
            const pin = document.getElementById('pin-input').value;
            if (pin === "1234" || pin === "") { // Default PIN: 1234
                document.getElementById('login-overlay').classList.add('hidden');
                document.getElementById('admin-content').classList.remove('hidden');
                document.getElementById('admin-store-name').textContent = store.name + " Admin";
                renderAdminItems();
                renderAdminCategories();
            } else {
                alert("Incorrect PIN code!");
            }
        }

        function saveStore() {
            localStorage.setItem('wh_store_data', JSON.stringify(store));
        }

        function resetDemoData() {
            if (confirm("Reset store data back to default demo items?")) {
                store = JSON.parse(JSON.stringify(DEFAULT_STORE));
                saveStore();
                location.reload();
            }
        }

        function switchTab(tab) {
            ['items', 'categories', 'settings'].forEach(t => {
                document.getElementById(`admin-tab-${t}`).classList.toggle('hidden', t !== tab);
                const btn = document.getElementById(`tab-btn-${t}`);
                btn.className = t === tab ? "py-2 text-emerald-600 border-b-2 border-emerald-600 font-semibold" : "py-2 hover:text-slate-700";
            });
            if (tab === 'settings') loadSettings();
        }

        function renderAdminItems() {
            const container = document.getElementById('admin-items-list');
            container.innerHTML = store.items.map(item => `
                <div class="bg-white p-4 rounded-xl border border-slate-200 flex justify-between items-center shadow-sm">
                    <div class="flex items-center gap-3">
                        ${item.image ? `<img src="${item.image}" class="w-12 h-12 rounded-lg object-cover">` : '<div class="w-12 h-12 bg-slate-100 rounded-lg flex items-center justify-center text-slate-400"><i class="fa-solid fa-utensils"></i></div>'}
                        <div>
                            <h4 class="font-bold text-slate-800">${item.name}</h4>
                            <div class="text-xs text-slate-500">${item.category} • ${store.currency}${item.price.toFixed(2)}</div>
                        </div>
                    </div>
                    <div class="flex gap-2">
                        <button onclick="openItemModal('${item.id}')" class="px-3 py-1.5 bg-slate-100 hover:bg-slate-200 text-slate-700 rounded-lg text-xs font-semibold">Edit</button>
                        <button onclick="deleteMenuItem('${item.id}')" class="px-3 py-1.5 bg-red-50 hover:bg-red-100 text-red-600 rounded-lg text-xs font-semibold">Delete</button>
                    </div>
                </div>
            `).join('');
        }

        function openItemModal(itemId = null) {
            const item = itemId ? store.items.find(i => i.id === itemId) : {
                id: Date.now().toString(),
                name: '',
                description: '',
                price: 0,
                category: store.categories[1] || 'Burgers',
                image: '',
                optionGroups: []
            };

            const modal = document.getElementById('item-modal-overlay');
            const content = document.getElementById('modal-content');

            content.innerHTML = `
                <div class="space-y-4">
                    <div class="flex justify-between items-center">
                        <h3 class="font-bold text-slate-900 text-lg">${itemId ? 'Edit Menu Item' : 'Add New Menu Item'}</h3>
                        <button onclick="closeModal()" class="text-slate-400 hover:text-slate-600"><i class="fa-solid fa-xmark text-xl"></i></button>
                    </div>

                    <form id="admin-item-form" onsubmit="saveMenuItem(event, '${item.id}')" class="space-y-3 text-sm">
                        <div>
                            <label class="block font-medium text-slate-700 mb-1">Item Name</label>
                            <input type="text" id="admin-item-name" value="${item.name}" required class="w-full px-3 py-2 border rounded-lg focus:ring-2 focus:ring-emerald-500 outline-none">
                        </div>

                        <div class="grid grid-cols-2 gap-3">
                            <div>
                                <label class="block font-medium text-slate-700 mb-1">Price (${store.currency})</label>
                                <input type="number" step="0.01" id="admin-item-price" value="${item.price}" required class="w-full px-3 py-2 border rounded-lg focus:ring-2 focus:ring-emerald-500 outline-none">
                            </div>
                            <div>
                                <label class="block font-medium text-slate-700 mb-1">Category</label>
                                <select id="admin-item-category" class="w-full px-3 py-2 border rounded-lg focus:ring-2 focus:ring-emerald-500 outline-none">
                                    ${store.categories.filter(c => c !== "All").map(c => `<option value="${c}" ${c === item.category ? 'selected' : ''}>${c}</option>`).join('')}
                                </select>
                            </div>
                        </div>

                        <div>
                            <label class="block font-medium text-slate-700 mb-1">Description</label>
                            <textarea id="admin-item-desc" rows="2" class="w-full px-3 py-2 border rounded-lg focus:ring-2 focus:ring-emerald-500 outline-none">${item.description}</textarea>
                        </div>

                        <div>
                            <label class="block font-medium text-slate-700 mb-1">Image URL</label>
                            <input type="url" id="admin-item-image" value="${item.image || ''}" placeholder="https://..." class="w-full px-3 py-2 border rounded-lg focus:ring-2 focus:ring-emerald-500 outline-none">
                        </div>

                        <div class="pt-3 flex justify-end gap-2">
                            <button type="button" onclick="closeModal()" class="px-4 py-2 border rounded-lg text-slate-600 hover:bg-slate-100">Cancel</button>
                            <button type="submit" class="px-4 py-2 bg-emerald-600 text-white font-semibold rounded-lg hover:bg-emerald-700">Save Item</button>
                        </div>
                    </form>
                </div>
            `;

            modal.classList.remove('hidden');
        }

        function closeModal() {
            document.getElementById('item-modal-overlay').classList.add('hidden');
        }

        function saveMenuItem(e, itemId) {
            e.preventDefault();
            const name = document.getElementById('admin-item-name').value.trim();
            const price = parseFloat(document.getElementById('admin-item-price').value);
            const category = document.getElementById('admin-item-category').value;
            const description = document.getElementById('admin-item-desc').value.trim();
            const image = document.getElementById('admin-item-image').value.trim();

            const existingIndex = store.items.findIndex(i => i.id === itemId);
            const itemObj = {
                id: itemId,
                name,
                price,
                category,
                description,
                image,
                optionGroups: existingIndex >= 0 ? store.items[existingIndex].optionGroups : []
            };

            if (existingIndex >= 0) {
                store.items[existingIndex] = itemObj;
            } else {
                store.items.push(itemObj);
            }

            saveStore();
            closeModal();
            renderAdminItems();
        }

        function deleteMenuItem(itemId) {
            if (confirm("Are you sure you want to delete this menu item?")) {
                store.items = store.items.filter(i => i.id !== itemId);
                saveStore();
                renderAdminItems();
            }
        }

        function renderAdminCategories() {
            const container = document.getElementById('admin-categories-list');
            container.innerHTML = store.categories.filter(c => c !== "All").map(cat => `
                <li class="p-3 flex justify-between items-center text-sm font-medium text-slate-700">
                    <span>${cat}</span>
                    <button onclick="deleteCategory('${cat}')" class="text-red-500 hover:underline text-xs">Remove</button>
                </li>
            `).join('');
        }

        function addCategory() {
            const input = document.getElementById('new-category-input');
            const val = input.value.trim();
            if (val && !store.categories.includes(val)) {
                store.categories.push(val);
                saveStore();
                input.value = '';
                renderAdminCategories();
            }
        }

        function deleteCategory(cat) {
            if (confirm(`Delete category "${cat}"?`)) {
                store.categories = store.categories.filter(c => c !== cat);
                saveStore();
                renderAdminCategories();
            }
        }

        function loadSettings() {
            document.getElementById('setting-name').value = store.name;
            document.getElementById('setting-phone').value = store.phone;
            document.getElementById('setting-currency').value = store.currency;
            document.getElementById('setting-greeting').value = store.greeting;
        }

        function saveSettings() {
            store.name = document.getElementById('setting-name').value.trim() || store.name;
            store.phone = document.getElementById('setting-phone').value.trim() || store.phone;
            store.currency = document.getElementById('setting-currency').value.trim() || store.currency;
            store.greeting = document.getElementById('setting-greeting').value.trim() || store.greeting;

            saveStore();
            alert("Settings saved successfully!");
        }
    </script>
